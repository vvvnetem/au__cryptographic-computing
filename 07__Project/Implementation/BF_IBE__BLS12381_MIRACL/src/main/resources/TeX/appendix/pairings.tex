\section{Equivalent transformations for pairings}
\label{sec:apx.equivtransform}

\begin{itemize}
	\item $e(P, S+T) = e(P,S) e(P(T))$, where $P \in \mathbb{G}_1, S,T \in \mathbb{G}_2$
	\item $e(aP, bQ) = e(bP, aQ)$,  where $a,b \in \mathbb{Z}_n$ and $P \in \mathbb{G}_1$, $Q \in \mathbb{G}_2$
	\item $e(aP, bQ) = e(abP,Q)$,  where $a,b \in \mathbb{Z}_n$ and $P \in \mathbb{G}_1$, $Q \in \mathbb{G}_2$
	\item $e(aP, bQ) = e(P,Q)^{ab}$,  where $a,b \in \mathbb{Z}_n$ and $P \in \mathbb{G}_1$, $Q \in \mathbb{G}_2$
	\item $e(-P, Q) = e(P,Q)^{-1} = e(P,-Q)$, where $P \in \mathbb{G}_1, Q \in \mathbb{G}_2$
	\item $e(P,0) = e(0, Q) = 1$ , where $P \in \mathbb{G}_1, Q \in \mathbb{G}_2$
\end{itemize}
